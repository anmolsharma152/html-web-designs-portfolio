// vad_processor: Client-side Rust WebAssembly VAD Binding
#[wasm_bindgen]
pub struct VadProcessor {
    session: ort::Session,
}

#[wasm_bindgen]
impl VadProcessor {
    pub fn is_speech(&mut self, pcm_chunk: &[f32]) -> Result {
        let prob = self.run_onnx_inference(pcm_chunk)?;
        Ok(prob > 0.5)
    }
}
